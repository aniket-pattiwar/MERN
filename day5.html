// restCrudServer.js - Complete Runnable RESTful API Server
const express = require('express');
const app = express();
const PORT = 5003;

app.use(express.json());

// In-memory data store for testing
let items = [
  { id: 1, name: 'Wireless Headphones', price: 99, category: 'electronics' },
  { id: 2, name: 'Ergonomic Chair', price: 249, category: 'furniture' },
  { id: 3, name: 'Mechanical Keyboard', price: 120, category: 'electronics' }
];

// 1. GET (Read All) with optional Query Param filtering: /api/items?category=electronics
app.get('/api/items', (req, res) => {
  const { category } = req.query;
  let results = items;
  if (category) {
    results = items.filter(i => i.category.toLowerCase() === category.toLowerCase());
  }
  res.status(200).json({ success: true, count: results.length, data: results });
});

// 2. GET (Read Single by Path Param): /api/items/1
app.get('/api/items/:id', (req, res) => {
  const item = items.find(i => i.id === parseInt(req.params.id));
  if (!item) return res.status(404).json({ success: false, error: 'Item not found' });
  res.status(200).json({ success: true, data: item });
});

// 3. POST (Create New Item with Request Body)
app.post('/api/items', (req, res) => {
  const { name, price, category } = req.body;
  if (!name || !price) {
    return res.status(400).json({ success: false, error: 'Name and price are required fields' });
  }
  const newItem = { id: Date.now(), name, price: Number(price), category: category || 'general' };
  items.push(newItem);
  res.status(201).json({ success: true, message: 'Item created!', data: newItem });
});

// 4. PUT (Update Existing Item by Path Param & Body)
app.put('/api/items/:id', (req, res) => {
  const item = items.find(i => i.id === parseInt(req.params.id));
  if (!item) return res.status(404).json({ success: false, error: 'Item not found' });

  item.name = req.body.name !== undefined ? req.body.name : item.name;
  item.price = req.body.price !== undefined ? Number(req.body.price) : item.price;
  item.category = req.body.category !== undefined ? req.body.category : item.category;

  res.status(200).json({ success: true, message: 'Item updated!', data: item });
});

// 5. DELETE (Remove Item by Path Param)
app.delete('/api/items/:id', (req, res) => {
  const index = items.findIndex(i => i.id === parseInt(req.params.id));
  if (index === -1) return res.status(404).json({ success: false, error: 'Item not found' });
  const deletedItem = items.splice(index, 1)[0];
  res.status(200).json({ success: true, message: 'Item deleted!', data: deletedItem });
});

app.listen(PORT, () => {
  console.log(`✅ REST API Server running at http://localhost:${PORT}`);
  console.log('👉 Try in browser: http://localhost:5003/api/items');
  console.log('👉 Try filter:     http://localhost:5003/api/items?category=electronics');
});
How it works: Implements full REST principles: GET for retrieval, POST (status 201) for creation, PUT for modification, and DELETE for removal, using both URL query params and route path params.

⭐ Understanding MVC Architecture (Model-View-Controller) in MERN

MVC is an architectural pattern that separates an application into three interconnected components, ensuring clean separation of concerns and maintainability:

Model (Data Layer)

Mongoose / MongoDB

  • Defines data structure, validation rules, and schema types.
  • Executes database queries (CRUD).
  • Does not know anything about the user interface.
View (Presentation Layer)

React.js (SPA)

  • Displays data to the user.
  • Captures user events (clicks, form inputs).
  • In decoupled MERN, React is the View consuming JSON APIs.
Controller (Logic Layer)

Express Route Handlers

  • Acts as the middleman between Model and View.
  • Processes incoming requests (req.body, req.params).
  • Calls Models/Repositories and sends JSON responses (res.json()).
How MVC Works in MERN Stack

💡 Simple Analogy A Restaurant Team (Kitchen, Waiter, Dining Table)

Think of MVC like a restaurant: Model is the kitchen and pantry (stores the ingredients & raw data in MongoDB); View is the dining table (the visual dish in React that the customer sees); and Controller is the waiter (takes the customer's request from the table, tells the kitchen what to prepare, and brings the food back).

⭐ Authentication & Authorization with JWT (JSON Web Tokens) & Protected Routes

Figure: Complete JWT Lifecycle (Login → Token Generation → LocalStorage → Auth Header → Verification)

Securing modern REST APIs requires stateless authentication so the server doesn't need to store session memory across distributed instances.

Authentication

"Who are you?"

Verifying user identity via credentials (e.g. Email & Password). Upon successful verification, the server issues a signed JWT token.

Authorization

"What are you allowed to do?"

Verifying user permissions/roles (e.g. admin vs user). Determining whether the authenticated user has access to a specific protected route or action.

1. Anatomy of a JSON Web Token (JWT)

A JWT is a string formatted as Header.Payload.Signature:

  • Header: Algorithm & token type (e.g., HMAC SHA256, JWT).
  • Payload: User claims & metadata (e.g., userId, role, expiration time exp). Never put sensitive passwords here.
  • Signature: Cryptographic hash created using server secret key to ensure the token hasn't been tampered with.

const mongoose = require('mongoose');
const bcrypt = require('bcryptjs');

const userSchema = new mongoose.Schema({
  name: { type: String, required: true },
  email: { type: String, required: true, unique: true },
  password: { type: String, required: true },
  role: { type: String, enum: ['user', 'admin'], default: 'user' }
});

// Pre-save hook to hash password before saving to MongoDB
userSchema.pre('save', async function (next) {
  if (!this.isModified('password')) return next();
  const salt = await bcrypt.genSalt(10);
  this.password = await bcrypt.hash(this.password, salt);
  next();
});

module.exports = mongoose.model('User', userSchema);
Step C: Auth Controller (`controllers/authController.js`)
const User = require('../models/User');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

const JWT_SECRET = process.env.JWT_SECRET || 'your_super_secret_fintech_key';

// 1. User Registration
exports.register = async (req, res) => {
  try {
    const { name, email, password, role } = req.body;
    const existingUser = await User.findOne({ email });
    if (existingUser) return res.status(400).json({ error: 'Email already exists' });

    const user = new User({ name, email, password, role });
    await user.save();
    res.status(201).json({ message: 'User registered successfully!' });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
};

// 2. User Login & Token Generation
exports.login = async (req, res) => {
  try {
    const { email, password } = req.body;
    const user = await User.findOne({ email });
    if (!user) return res.status(401).json({ error: 'Invalid credentials' });

    // Compare entered password with hashed password in DB
    const isMatch = await bcrypt.compare(password, user.password);
    if (!isMatch) return res.status(401).json({ error: 'Invalid credentials' });

    // Generate signed JWT token
    const token = jwt.sign(
      { id: user._id, role: user.role, name: user.name },
      JWT_SECRET,
      { expiresIn: '2h' }
    );

    res.json({
      message: 'Login successful',
      token: token,
      user: { id: user._id, name: user.name, role: user.role }
    });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
};
Step D: JWT Authentication Middleware (`middleware/authMiddleware.js`)
const jwt = require('jsonwebtoken');
const JWT_SECRET = process.env.JWT_SECRET || 'your_super_secret_fintech_key';

// Middleware to verify JWT token in Authorization header
const authenticateJWT = (req, res, next) => {
  const authHeader = req.headers.authorization;

  if (authHeader && authHeader.startsWith('Bearer ')) {
    const token = authHeader.split(' ')[1];

    jwt.verify(token, JWT_SECRET, (err, decodedUser) => {
      if (err) {
        return res.status(403).json({ error: 'Token is invalid or expired' });
      }
      req.user = decodedUser; // Attach user payload to request
      next(); // Proceed to route controller
    });
  } else {
    res.status(401).json({ error: 'Access denied: No token provided' });
  }
};

// Optional Role Authorization Middleware (e.g. admin only)
const authorizeRole = (role) => {
  return (req, res, next) => {
    if (req.user.role !== role) {
      return res.status(403).json({ error: 'Forbidden: Insufficient permissions' });
    }
    next();
  };
};

module.exports = { authenticateJWT, authorizeRole };

// jwtAuthServer.js - Complete JWT Token Creation & Protected Route Middleware
const express = require('express');
const jwt = require('jsonwebtoken');

const app = express();
const PORT = 5004;
const JWT_SECRET = 'my_super_secret_jwt_key_123'; // In production, keep in .env

app.use(express.json());

// 1. Login Route: Issue a JWT upon credentials check
app.post('/api/login', (req, res) => {
  const { username, password } = req.body;

  // Simple credential verification (in production, verify with bcrypt + MongoDB)
  if (username === 'student' && password === 'mern123') {
    const payload = { id: 101, username: 'student', role: 'admin' };
    
    // Sign token (expires in 1 hour)
    const token = jwt.sign(payload, JWT_SECRET, { expiresIn: '1h' });
    return res.json({ success: true, message: 'Authentication successful!', token });
  }

  res.status(401).json({ success: false, error: 'Invalid username or password' });
});

// 2. Authentication Middleware: Verify Bearer Token
const verifyToken = (req, res, next) => {
  const authHeader = req.headers['authorization']; // Expecting "Bearer <token>"
  const token = authHeader && authHeader.split(' ')[1];

  if (!token) {
    return res.status(401).json({ error: 'Access Denied: No Bearer token provided in Authorization header.' });
  }

  jwt.verify(token, JWT_SECRET, (err, decodedUser) => {
    if (err) {
      return res.status(403).json({ error: 'Forbidden: Invalid or expired token.' });
    }
    // Attach verified user payload to the request object
    req.user = decodedUser;
    next();
  });
};

// 3. Protected Route: Only accessible with a valid JWT
app.get('/api/protected/profile', verifyToken, (req, res) => {
  res.json({
    message: 'Welcome to your protected profile!',
    user: req.user,
    secretData: 'Confidential grade records: GPA 4.0'
  });
});

app.listen(PORT, () => {
  console.log(`✅ JWT Auth Server running at http://localhost:${PORT}`);
  console.log('👉 1. Login with POST /api/login {"username": "student", "password": "mern123"}');
  console.log('👉 2. Send GET /api/protected/profile with Header: Authorization: Bearer <token>');
});
How it works: Demonstrates stateless token-based security. jwt.sign() issues a tamper-proof cryptographically signed token. The verifyToken middleware extracts the Bearer token from the Authorization header and verifies the signature using jwt.verify().
